{{- /* Rewrites relative ".md" links (e.g. "api.md", "../weighting/kernels.md#foo") into the linked page's real permalink via .Page.GetPage, which resolves the path relative to the current page's content-tree location. This means link targets resolve correctly regardless of category-folder nesting depth, without any manual "../" climbing. */ -}} {{- $dest := .Destination -}} {{- if strings.Contains $dest ".md" -}} {{- $anchor := "" -}} {{- $path := $dest -}} {{- if strings.Contains $dest "#" -}} {{- $parts := split $dest "#" -}} {{- $path = index $parts 0 -}} {{- $anchor = printf "#%s" (index $parts 1) -}} {{- end -}} {{- with .Page.GetPage $path -}} {{- $dest = printf "%s%s" .RelPermalink $anchor -}} {{- end -}} {{- end -}} {{ .Text | safeHTML }}
