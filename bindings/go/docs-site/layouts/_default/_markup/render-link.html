{{- /* Rewrites bare same-directory ".md" links (e.g. "api.md", "api.md#custom-weights") so they resolve correctly under pretty URLs, where every content page gets its own directory (e.g. content/kernels.md -> /kernels/). A same-directory link therefore has to climb back out of the current page's own directory first. */ -}} {{- $dest := .Destination -}} {{- if strings.Contains $dest ".md" -}} {{- if strings.Contains $dest "#" -}} {{- $parts := split $dest "#" -}} {{- $file := strings.TrimSuffix ".md" (index $parts 0) -}} {{- $dest = printf "../%s/#%s" $file (index $parts 1) -}} {{- else -}} {{- $file := strings.TrimSuffix ".md" $dest -}} {{- $dest = printf "../%s/" $file -}} {{- end -}} {{- end -}} {{ .Text | safeHTML }}
