59template <typename T> class Expected {
60public:
61 // Success constructor
62 explicit Expected(T val) : val_(std::move(val)), has_val_(true) {}
63
64 // Error constructor
65 struct ErrorTag {};
66 static Expected make_error(std::string msg) {
67 return Expected(std::move(msg), ErrorTag{});
68 }
69
70 bool has_value() const { return has_val_; }
71
72 explicit operator bool() const { return has_val_; }
73
74 T &value() & {
75 if (!has_val_) {
76 throw LowessError(err_);
77 }
78 return val_;
79 }
80
81 const T &value() const & {
82 if (!has_val_) {
83 throw LowessError(err_);
84 }
85 return val_;
86 }
87
88 T &&value() && {
89 if (!has_val_) {
90 throw LowessError(err_);
91 }
92 return std::move(val_);
93 }
94
95 const std::string &error() const {
96 if (has_val_) {
97 throw LowessError("Bad expected access: has value");
98 }
99 return err_;
100 }
101
102private:
103 Expected(std::string err, ErrorTag error_tag)
104 : err_(std::move(err)), has_val_(false) {
105 static_cast<void>(error_tag);
106 }
107
108 // We store both to avoid manual union management, relying on T's cheap
109 // default ctor. LowessResult's default ctor is cheap (zero-init).
110 T val_;
111 std::string err_;
112 bool has_val_;
113};

733 if (available()) {
734 std::cout << "GPU backend is already active.\n";

736 }
737
738 const std::string version = cpp_version();
739 const std::string asset = "libfastlowess_cpp-gpu-v" + version + "-" +
740 detail::platformTag() + "-" + detail::archTag() +
741 "." + detail::libraryExt();
742 const std::string url =
743 "https://github.com/thisisamirv/lowess-project/releases/download/v" +
744 version + "/" + asset;
745
746 if (!yes) {
747 if (!detail::stdinIsInteractive()) {
748 std::cerr << "install_gpu() requires confirmation; pass yes=true to "
749 "proceed non-interactively."
750 << "\n";
751 return false;
752 }
753 std::cout << "Download and install " << asset
754 << " from github.com/thisisamirv/lowess-project? [y/N] ";
755 std::string answer;
756 std::getline(std::cin, answer);
757 if (answer != "y" && answer != "Y" && answer != "yes") {
758 std::cout << "Aborted.\n";
759 return false;
760 }
761 }
762
763 const std::string dir = detail::cacheDir();
764 detail::makeDir(dir);
765 const std::string dest = dir + "/" + asset;
766
767 std::cout << "Downloading " << url << " ...\n";
768 const std::string curl_cmd =
769 "curl -fL --progress-bar -o \"" + dest + "\" \"" + url + "\"";
770 if (std::system(curl_cmd.c_str()) != 0) { // NOLINT
771 std::cerr << "Download failed. A matching GPU build may not exist for "
772 "this platform/version yet, or `curl` is not installed."
773 << "\n";
774 return false;
775 }
776
777 std::cout << "Downloaded to " << dest << "\n";
778 std::cout << "Relink your application against this library (or "
779 "dlopen/LoadLibrary it) and restart to use the GPU backend."
780 << "\n";
781 return true;
782}
