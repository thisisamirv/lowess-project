60template <typename T> class Expected {
61public:
62 // Success constructor
63 explicit Expected(T val) : val_(std::move(val)), has_val_(true) {}
64
65 // Error constructor
66 struct ErrorTag {};
67 static Expected make_error(std::string msg) {
68 return Expected(std::move(msg), ErrorTag{});
69 }
70
71 bool has_value() const { return has_val_; }
72
73 explicit operator bool() const { return has_val_; }
74
75 T &value() & {
76 if (!has_val_) {
77 throw LowessError(err_);
78 }
79 return val_;
80 }
81
82 const T &value() const & {
83 if (!has_val_) {
84 throw LowessError(err_);
85 }
86 return val_;
87 }
88
89 T &&value() && {
90 if (!has_val_) {
91 throw LowessError(err_);
92 }
93 return std::move(val_);
94 }
95
96 const std::string &error() const {
97 if (has_val_) {
98 throw LowessError("Bad expected access: has value");
99 }
100 return err_;
101 }
102
103private:
104 Expected(std::string err, ErrorTag error_tag)
105 : err_(std::move(err)), has_val_(false) {
106 static_cast<void>(error_tag);
107 }
108
109 // We store both to avoid manual union management, relying on T's cheap
110 // default ctor. LowessResult's default ctor is cheap (zero-init).
111 T val_;
112 std::string err_;
113 bool has_val_;
114};

478 Expected<LowessResult> fit(const std::vector<double> &x_values,

491 cpp_lowess_fit(ptr_, x_values.data(), y_values.data(),
492 static_cast<unsigned long>(x_values.size()),
493 custom_weights.empty() ? nullptr : custom_weights.data(),
494 static_cast<unsigned long>(custom_weights.size()));
495
496 if (result.error != nullptr) {
497 const std::string error_msg(result.error);
498 cpp_lowess_free_result(&result);
