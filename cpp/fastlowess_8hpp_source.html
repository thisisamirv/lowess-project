1029inline bool install(bool yes = false, const std::string &local_path = "") {
1030 if (available()) {
1031 std::cout << "GPU backend is already active.\n";
1032 return true;
1033 }
1034
1035 const std::string dir = detail::cacheDir();
1036 detail::makeDir(dir);
1037
1038 if (!local_path.empty()) {
1039 if (!detail::fileExists(local_path)) {
1040 std::cerr << "No such file: " << local_path << "\n";
1041 return false;
1042 }
1043 if (!yes) {
1044 if (!detail::stdinIsInteractive()) {
1045 std::cerr << "install_gpu() requires confirmation; pass yes=true to "
1046 "proceed non-interactively."
1047 << "\n";
1048 return false;
1049 }
1050 std::cout << "Install " << local_path
1051 << " in place of the current build? [y/N] ";
1052 std::string answer;
1053 std::getline(std::cin, answer);
1054 if (answer != "y" && answer != "Y" && answer != "yes") {
1055 std::cout << "Aborted.\n";
1056 return false;
1057 }
1058 }
1059
1060 const std::string dest =
1061 dir + "/" + local_path.substr(local_path.find_last_of("/\\") + 1);
1062 std::cout << "Installing " << local_path << " ...\n";
1063 if (!detail::copyFile(local_path, dest)) {
1064 std::cerr << "Failed to copy " << local_path << " to " << dest << "\n";
1065 return false;
1066 }
1067 std::cout << "Installed to " << dest << "\n";
1068 std::cout << "Relink your application against this library (or "
1069 "dlopen/LoadLibrary it) and restart to use the GPU backend."
1070 << "\n";
1071 return true;
1072 }
1073
1074 const std::string version = cpp_version();
1075 const std::string asset = "libfastlowess_cpp-gpu-v" + version + "-" +
1076 detail::platformTag() + "-" + detail::archTag() +
1077 "." + detail::libraryExt();
1078 // GPU artifacts across all versions live in this one perpetual release
1079 // instead of cluttering each version's own release page; the source
1080 // version is embedded in the asset filename above instead.
1081 const std::string url =
1082 "https://github.com/thisisamirv/lowess-project/releases/download/"
1083 "gpu-builds/" +
1084 asset;
1085
1086 if (!yes) {
1087 if (!detail::stdinIsInteractive()) {
1088 std::cerr << "install_gpu() requires confirmation; pass yes=true to "
1089 "proceed non-interactively."
1090 << "\n";
1091 return false;
1092 }
1093 std::cout << "Download and install " << asset
1094 << " from github.com/thisisamirv/lowess-project? [y/N] ";
1095 std::string answer;
1096 std::getline(std::cin, answer);
1097 if (answer != "y" && answer != "Y" && answer != "yes") {
1098 std::cout << "Aborted.\n";
1099 return false;
1100 }
1101 }
1102
1103 const std::string dest = dir + "/" + asset;
1104
1105 std::cout << "Downloading " << url << " ...\n";
1106 const std::string curl_cmd =
1107 "curl -fL --progress-bar -o \"" + dest + "\" \"" + url + "\"";
1108 if (std::system(curl_cmd.c_str()) != 0) { // NOLINT
1109 std::cerr << "Download failed. A matching GPU build may not exist for "
1110 "this platform/version yet, or `curl` is not installed."
1111 << "\n";
1112 return false;
1113 }
1114
1115 std::cout << "Downloaded to " << dest << "\n";
1116 std::cout << "Relink your application against this library (or "
1117 "dlopen/LoadLibrary it) and restart to use the GPU backend."
1118 << "\n";
1119 return true;
1120}
